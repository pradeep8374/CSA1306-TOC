#include <stdio.h>
#include <string.h>

int main() {
    char s[100];
    int i = 0, zeros = 0, ones = 0, valid = 1;

    printf("Enter a binary string: ");
    scanf("%99s", s);

    while (s[i] == '0') {
        zeros++;
        i++;
    }

    while (s[i] == '1') {
        ones++;
        i++;
    }

    if (s[i] != '\0' || zeros != ones)
        valid = 0;

    if (valid)
        printf("String belongs to the grammar\n");
    else
        printf("String does not belong to the grammar\n");

    printf("Press Enter to close...");
    getchar();
    getchar();

    return 0;
}
